# Import libraries
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Load Iris dataset (inbuilt)
df = sns.load_dataset('iris')

# Display dataset
print(df.head())
print(df.info())

# 1. Features and their types
print("\nFeature Types:")
print(df.dtypes)

# 2. Histogram for each feature
df.hist(figsize=(10,8))
plt.suptitle("Histogram of Features")
plt.show()

# 3. Boxplot for each feature
plt.figure(figsize=(10,8))

# Melt dataset for easier plotting
df_melt = df.melt(id_vars='species')

sns.boxplot(x='variable', y='value', data=df_melt)

plt.title("Boxplot of Features")
plt.show()
