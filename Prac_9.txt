# Import libraries
import seaborn as sns
import matplotlib.pyplot as plt

# Load dataset
df = sns.load_dataset('titanic')

# Display dataset
print(df.head())
print(df.info())

# Boxplot: Age distribution w.r.t gender and survival
sns.boxplot(x='sex', y='age', hue='survived', data=df)

# Title and labels
plt.title("Age Distribution by Gender and Survival")
plt.xlabel("Gender")
plt.ylabel("Age")

plt.show()
